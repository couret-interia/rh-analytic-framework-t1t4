\section*{Notation summary (quick reference)}
\begin{tabular}{@{}ll@{}}
$\Lambda(n)$ & von Mangoldt function \\[2pt]
$\rho=\tfrac12+i\gamma$ & nontrivial zero of $\zeta(s)$ \\[2pt]
$\PW$ & Paley--Wiener class (even, compact support $\subset[-A,A]$) \\[2pt]
$\Xi_A$ & logarithmic sampling grid $\{\,k\log p\le A\,\}$ \\[2pt]
$w_{p,k}$ & arithmetic weights $\log p\,/\,p^{k/2}$ \\[2pt]
$G^{(w)}$ & weighted Gram (frame) matrix \\[2pt]
$\lambda_{\min}(G^{(w)})$ & smallest eigenvalue (frame gap) \\[2pt]
$\Cfrw(A)$ & frame constant $1-\Theta_w(A)$ \\[2pt]
$\phi_T,\ \hat\phi_T$ & wave packet and its Fourier transform \\[2pt]
$K(t)$ & Archimedean term (from $\Gamma'/\Gamma$)
\end{tabular}
